## kubernetes Task - Create Nginx provider
apiVersion: V1
kind: Pod 
metadata:
   name: pathnex-pod 
spec:
   containers:
     -name: web
     image: nginx:latest
     ports:
       - containerPort: 80

#Git Integration
## Jenkins Pipeline - checkout Git

pipeline {
    agent any
    stages {
        stage('checkout'){
            steps {
                git branch: 'main' , url: 'https://github.com/Pathnex/sample-java-app.git'
        }
    }
    stage('List files') {
        steps {
            sh 'ls -la'
           }
        }
    }
}
