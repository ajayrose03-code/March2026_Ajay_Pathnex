## Jenkins Pipeline - Archive Artifacts


pipline  {
   agent any
   environment {
       INSTITUE_NAME = "APthnex"
    }
    stages   {
       stage ('Checkout'){
           steps  {
	        git url: 'https://github.com/pathnex/sample-java-app.git'
	}
      }
      stage ('Build'){
         steps  {
	      sh 'mvn clean package'
	   }
	}
	stage ('Archive'){
	  steps {
	        achivefacts artifacts: 'target/*.jar', fingerprint: true
	       }
	    }
	  }
      }
