# Multi-Branch Pipeline
## Jenkins pipeline- Multi-Branch

pipeline {
        agent any
        environment {
            INSTITUE_NAME = "Pathnex"
        }
        stages {
                stage ('Checkout') {
                        steps {
                            checkout scm
                        }
                }
                stage ('Build') {
                    steps {
                        sh 'mvn clean compile'
                    }
                }
        }
}
